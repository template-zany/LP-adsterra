  <script>
    function randomViewers() {
      const v = Math.floor(Math.random() * 9000) + 1000;
      document.getElementById('viewerCount').textContent = v.toLocaleString();
    }

    function updateMatchStatus() {
      const nowUTC = new Date();
      const utcMinutes = nowUTC.getUTCHours() * 60 + nowUTC.getUTCMinutes();
      const wibMinutes = utcMinutes + 420;

      document.querySelectorAll('#jadwalList li').forEach(li => {
        const timeStr = li.getAttribute('data-time');
        const [h, m] = timeStr.split(':').map(Number);
        const matchMinutes = h * 60 + m;
        const label = li.querySelector('.status-label');
        label.className = 'status-label';

        if (wibMinutes >= matchMinutes && wibMinutes <= matchMinutes + 120) {
          label.textContent = 'Live Sekarang';
          label.classList.add('status-live');
        } else if (wibMinutes < matchMinutes) {
          label.textContent = 'Segera Live';
          label.classList.add('status-soon');
        } else {
          label.textContent = 'Selesai';
          label.classList.add('status-finished');
        }
      });
    }

    window.onload = () => {
      randomViewers();
      updateMatchStatus();
      setInterval(randomViewers, 5000);
      setInterval(updateMatchStatus, 60000);
    };
  </script>
